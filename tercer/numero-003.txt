#!/usr/bin/env node
/* Lo sistema no sent dolor, només executa la instrucció.
   Ús: node virtual-ausias.js [--traduir] */

"use strict";

const W = 40;
const TEXT = W - 2;
const TRADUIR = process.argv.includes("--traduir");

/* Versos recuperats. */
const VERSOS = [
  "Callen aquells      que d’Amor han parlat,",
  "e dels passats      deliu tots lurs escrits,",
  "e·n mi penssant     meteu-los en oblits.",
  "Lexant a part       l'estil dels trobadors",
  "qui, per escalf,    trespassen veritat",
  "No guart avant ne membre lo passat",
];

/* Traducció de treball. */
const TRADUCCIO = [
  "Callin aquells      que d'Amor han parlat,",
  "i dels passats      esborreu tots llurs escrits,",
  "i pensant en mi     poseu-los en l'oblit.",
  "Deixant de banda    l'estil dels trovadors",
  "qui, per escalf,    traspassen la veritat",
  "No miro endavant ni recordo el passat",
];

/* Extracció: text → ordre. */
const REGLES = [
  { vers: /callen aquells/i,      ordre: "SILENCIAR" },
  { vers: /deliu tots lurs/i,     ordre: "DELIU" },
  { vers: /meteu-los en oblits/i, ordre: "OBLIT" },
  { vers: /lexant a part/i,       ordre: "LEXANT" },
  { vers: /trespassen veritat/i,  ordre: "TRESPASSAR" },
  { vers: /no guart avant/i,      ordre: "NO GUART" },
];

function compila() {
  return VERSOS.map(function (vers) {
    const regla = REGLES.find(function (r) { return r.vers.test(vers); });
    if (!regla) throw new Error("Vers sense ordre: " + vers);
    return { vers: vers, ordre: regla.ordre };
  });
}

/* Escriptura a 40 columnes. */
function out(text) {
  if (!text) {
    console.log("█" + " ".repeat(W - 1));
    return;
  }
  const paraules = text.split(" ");
  let linia = "";
  paraules.forEach(function (p) {
    if (linia.length + (linia ? 1 : 0) + p.length <= TEXT) {
      linia += (linia ? " " : "") + p;
    } else {
      console.log("█ " + linia.padEnd(TEXT, " "));
      linia = p;
    }
  });
  if (linia) console.log("█ " + linia.padEnd(TEXT, " "));
}

function outRaw(text) {
  if (!text) {
    console.log("█" + " ".repeat(W - 1));
    return;
  }
  let linia = "";
  const flush = function () {
    console.log("█ " + linia.padEnd(TEXT, " "));
    linia = "";
  };
  text.split(/( +)/).forEach(function (part) {
    if (!part) return;
    if (/^ +$/.test(part)) {
      if (linia.length + part.length <= TEXT) {
        linia += part;
      } else {
        flush();
      }
      return;
    }
    if (linia.length + part.length <= TEXT) {
      linia += part;
      return;
    }
    if (linia) flush();
    while (part.length > TEXT) {
      console.log("█ " + part.slice(0, TEXT).padEnd(TEXT, " "));
      part = part.slice(TEXT);
    }
    linia = part;
  });
  if (linia) flush();
}

/* --- Execució ---------------------------------------------- */

console.log("█" + "▄".repeat(W - 1));
out("Lo sistema no sent dolor, només");
out("executa la instrucció.");
out("");

const pla = compila();

pla.forEach(function (pas, i) {
  out(String(i + 1).padStart(2, "0") + " · " + pas.ordre);
  outRaw(pas.vers);
  if (TRADUIR) out("(«" + TRADUCCIO[i] + "»)");
  out("");
});

out("FI. CAP VEU. CAP ESCRIT. CAP");
out("PASSAT. NOMÉS TERME, NOMÉS CORS,");
out("NOMÉS CARN, NOMÉS BYTES.");
out("NOMÉS AUSIÀS MARCH.");
console.log("█" + "▄".repeat(W - 1));
